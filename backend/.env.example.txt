# =============================================================
# Bravo Music API — Environment Configuration
# =============================================================
# Copy this file to .env and fill in the actual values for your
# environment. NEVER commit .env to git.
#
# Required vars cause the server to refuse to boot if missing.
# Recommended vars produce warnings but are not blocking.
#
# All values shown are PLACEHOLDERS — the validateEnv check in
# production refuses to start if the value contains 'changeme',
# 'example', 'your-', 'placeholder', or 'todo'.
# =============================================================

# ----- Core (REQUIRED everywhere) -----

# 'development' | 'staging' | 'production' | 'test'
NODE_ENV=development

# MongoDB connection URI (mongodb:// or mongodb+srv://)
MONGO_URI=mongodb://localhost:27017/bravomusic

# JWT signing secrets — generate via: openssl rand -base64 48
# Must be DIFFERENT for access vs refresh tokens.
JWT_SECRET=changeme-32-or-more-chars-of-random-data
JWT_REFRESH_SECRET=changeme-DIFFERENT-32-or-more-chars

# Token lifetimes (defaults shown). Used by authController.
JWT_EXPIRES_IN=15m
JWT_REFRESH_EXPIRES_IN=30d


# ----- Network (REQUIRED in production) -----

# Server listen port (default 1000)
PORT=1000

# Comma-separated CORS allowlist. NO localhost in production.
# Example: ALLOWED_ORIGINS=https://bravomusics.com,https://www.bravomusics.com
ALLOWED_ORIGINS=https://bravomusics.com,https://www.bravomusics.com,https://api.bravomusics.com

# Public frontend URL (used in emails for reset / verify links)
FRONTEND_URL=https://bravomusics.com


# ----- PawaPay (REQUIRED in production) -----

# Get these from your PawaPay dashboard.
PAWAPAY_API_KEY=changeme
PAWAPAY_WEBHOOK_SECRET=changeme-min-16-chars
PAWAPAY_BASE_URL=https://api.pawapay.io

# Set to 'sandbox' to use PawaPay's test environment.
# Defaults to 'production' if NODE_ENV=production, else 'sandbox'.
PAWAPAY_ENV=sandbox


# ----- Email / SMTP (RECOMMENDED) -----

# If unset, password reset / verify / payment receipts won't be sent.
EMAIL_FROM=no-reply@bravomusics.com
EMAIL_SMTP_HOST=smtp.example.com
EMAIL_SMTP_PORT=587
EMAIL_SMTP_USER=changeme
EMAIL_SMTP_PASS=changeme
EMAIL_SMTP_SECURE=true


# ----- Storage paths (RECOMMENDED) -----

# Where multer writes uploaded files. If unset, defaults to ./uploads.
# In production, mount a persistent volume here.
STATIC_UPLOADS_PATH=/var/www/bravo/uploads

# Where the legacy static music + images live (if you still use them).
STATIC_MUSIC_PATH=/var/www/bravo/static/music
STATIC_IMAGES_PATH=/var/www/bravo/static/images


# ----- AWS S3 (OPTIONAL — for cloud storage instead of disk) -----

# AWS_REGION=eu-west-1


# ----- Backups (RECOMMENDED for production) -----

# If unset, the backupService writes to /tmp (bad in prod).
BACKUP_S3_BUCKET=bravo-music-backups
BACKUP_RETENTION_DAYS=30
